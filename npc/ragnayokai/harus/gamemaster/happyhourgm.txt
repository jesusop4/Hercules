sec_in02,20,168,5	script	Happy Hour[GM]	728,{
			
		if (getgmlevel() != 99) end;
		
		mes "[Bônus Campal]";
		mes "Olá administrador!";
		mes "O que você deseja fazer?";	
		next;
		switch(select("Ativar Bônus:Desativar Bônus"))
		{
			case 1:
				if (getbattleflag("bg_ranked_mode") != 0)
				{
					mes "[Bônus Campal]";	
					mes "O Bônus já está ativado!!";
					close;

				}
				mes "[Bônus Campal]";
				mes "Qual será a taxa de bônus?(1..100)";
				next;
				input .@rate;
				if (.@rate == 0 || .@rate > 100 ) close;
				next;
				mes "[Bônus Campal]";
				mes "O bônus será rate + " + .@rate + ".";
				mes "Você está certo disto?";
				mes "PS: Bônus oficial é: 30%";
				next;
				if(select("Não:Sim")==1)close;
				announce "" + strcharinfo(0) + " - Ativou o Happy Hour da Batalha Campal. + " + .@rate + "% de bônificação!!!",0,0x00FF00;
				setbattleflag "bg_reward_rates",(100+.@rate); // +20% de Bonus padrão
				setbattleflag "bg_ranked_mode",1;
				break;
		
			case 2:
				announce "" + strcharinfo(0) + " - Desativou o Happy Hour da Batalha Campal.",0,0x00BFFF;
				setbattleflag "bg_reward_rates",100; // Rates Padrão
				setbattleflag "bg_ranked_mode",0;
				break;

		}
		close;

}
