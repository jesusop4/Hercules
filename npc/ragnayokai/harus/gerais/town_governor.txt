//==============================================================
// Town Governor — Town Donation & Buff System
//
// Towns: 0=Prontera 1=Morocc 2=Geffen 3=Payon 4=Alberta
//        5=Izlude 6=Aldebaran 7=Lutie 8=Comodo 9=Yuno
//        10=Amatsu 11=Gonryun 12=Louyang 13=Hugel
//
// Persistent variables per char:
//   TG_DONATED_<TOWN>  — total zeny donated to each town
//   TG_BUFF_TOWN       — town_id+1 of current buff (0=none)
//
// Global NPC variables:
//   $TG_TOP_ID[14]     — char_id of top donor per town
//   $TG_TOP_NAME$[14]  — name of top donor per town
//   $TG_TOP_ZEN[14]    — zeny donated by top donor
//
// Weekly reset: Monday 00:01
//==============================================================

//--------------------------------------------------------------
// F_TG_ApplyBuff(town_id) — apply the buff for a town
//--------------------------------------------------------------
function	script	F_TG_ApplyBuff	{
	set @_tg_town, getarg(0);
	set @_tg_dur, 604800000; // 7 days in ms

	if (@_tg_town == 0) { // Prontera
		sc_start SC_INCMHPRATE,@_tg_dur,10,0;
		sc_start SC_INCMSPRATE,@_tg_dur,5,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
		sc_start SC_INCALLSTATUS,@_tg_dur,3,0;
	} else if (@_tg_town == 1) { // Morocc
		sc_start SC_INCAGI,@_tg_dur,5,0;
		sc_start SC_INCSTR,@_tg_dur,3,0;
		sc_start SC_INCLUK,@_tg_dur,3,0;
		sc_start SC_INCFLEE,@_tg_dur,10,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 2) { // Geffen
		sc_start SC_INCINT,@_tg_dur,3,0;
		sc_start SC_INCDEX,@_tg_dur,3,0;
		sc_start SC_INCVIT,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 3) { // Payon
		sc_start SC_INCAGI,@_tg_dur,2,0;
		sc_start SC_INCDEX,@_tg_dur,3,0;
		sc_start SC_INCLUK,@_tg_dur,3,0;
		sc_start SC_INCFLEE,@_tg_dur,10,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 4) { // Alberta
		sc_start SC_INCSTR,@_tg_dur,5,0;
		sc_start SC_INCAGI,@_tg_dur,2,0;
		sc_start SC_INCVIT,@_tg_dur,3,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 5) { // Izlude
		sc_start SC_INCVIT,@_tg_dur,5,0;
		sc_start SC_INCMSPRATE,@_tg_dur,5,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 6) { // Aldebaran
		sc_start SC_INCAGI,@_tg_dur,3,0;
		sc_start SC_INCINT,@_tg_dur,3,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
		sc_start SC_SPEEDUP0,@_tg_dur,1,0;
	} else if (@_tg_town == 7) { // Lutie
		sc_start SC_INCLUK,@_tg_dur,3,0;
		sc_start SC_INCSTR,@_tg_dur,3,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 8) { // Comodo
		sc_start SC_INCAGI,@_tg_dur,3,0;
		sc_start SC_INCLUK,@_tg_dur,3,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 9) { // Yuno
		sc_start SC_INCINT,@_tg_dur,3,0;
		sc_start SC_INCVIT,@_tg_dur,3,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 10) { // Amatsu
		sc_start SC_INCDEX,@_tg_dur,3,0;
		sc_start SC_INCINT,@_tg_dur,3,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;
	} else if (@_tg_town == 11) { // Gonryun
		sc_start SC_INCVIT,@_tg_dur,3,0;
		sc_start SC_INCINT,@_tg_dur,3,0;
		sc_start SC_INCATKRATE,@_tg_dur,5,0;
		sc_start SC_INCMATKRATE,@_tg_dur,5,0;

//==============================================================
// Main Governor NPC (invisible, triggered by each town NPC)
//==============================================================
-	script	TownGovernorSystem	-1,{

OnPCLoginEvent:
	// Reapply buff if this character is still the top donor
	if (TG_BUFF_TOWN <= 0) end;
	set @_tg_t, TG_BUFF_TOWN - 1;
	if ($TG_TOP_ID[@_tg_t] != getcharid(0)) {
		// No longer top donor — remove buff record
		set TG_BUFF_TOWN, 0;
		end;
	}
	// Still top donor — reapply buff
	callfunc "F_TG_ApplyBuff", @_tg_t;
	end;

OnTalk:
	// Called by individual town governor NPCs with setarg(0, town_id)
	set @_tg_town, @tg_town_id;
	set @_tg_name$, callfunc("F_TG_TownName", @_tg_town);

L_Main:
	mes "^663300[ Town Governor ]^000000";
	mes "^AA6600" + @_tg_name$ + " Donation Fund^000000";
	mes " ";
	if ($TG_TOP_ID[@_tg_town] > 0) {
		mes "Top Donor: ^FF8800" + $TG_TOP_NAME$[@_tg_town] + "^000000";
		mes "Donated:   ^FF8800" + $TG_TOP_ZEN[@_tg_town] + " z^000000";
	} else {
		mes "^888888No donations yet. Be the first!^000000";
	}
	mes " ";
	mes "Your donation: ^FF8800" + callfunc("F_TG_GetDonated", @_tg_town) + " z^000000";
	if (TG_BUFF_TOWN == @_tg_town + 1) {
		mes "Status: ^00AA00You have the Top Donor buff!^000000";
	}
	mes " ";
	mes "^888888Donation Buff:^000000";
	callfunc "F_TG_ShowBuff", @_tg_town;
	next;
	set @_tg_sel, select("Donate Zeny", "View Leaderboard", "Close");
	if (@_tg_sel == 1) goto L_Donate;
	if (@_tg_sel == 2) goto L_Board;
	end;

L_Donate:
	mes "^663300[ Donate to " + @_tg_name$ + " ]^000000";
	mes "Enter the amount to donate (min 10,000z):";
	mes "Your Zeny: ^FF8800" + Zeny + " z^000000";
	next;
	input @_tg_amount, 10000, 1000000000;
	if (@_tg_amount % 10000 != 0) {
		mes "^663300[ Donate to " + @_tg_name$ + " ]^000000";
		mes "^FF0000Amount must be a multiple of 10,000z.^000000";
		next;
		goto L_Donate;
	}
	if (Zeny < @_tg_amount) {
		mes "^663300[ Donate to " + @_tg_name$ + " ]^000000";
		mes "^FF0000You don't have enough Zeny.^000000";
		next;
		goto L_Main;
	}
	set Zeny, Zeny - @_tg_amount;
	callfunc "F_TG_AddDonated", @_tg_town, @_tg_amount;
	set @_tg_my_total, callfunc("F_TG_GetDonated", @_tg_town);

	// Check if new top donor
	if (@_tg_my_total > $TG_TOP_ZEN[@_tg_town]) {
		// Remove buff from previous top donor if online
		if ($TG_TOP_ID[@_tg_town] > 0 && $TG_TOP_ID[@_tg_town] != getcharid(0)) {
			set @_tg_prev_sd, $TG_TOP_ID[@_tg_town];
			// Will be removed on their next login via OnPCLoginEvent check
		}
		// Remove my own previous buff (another town) if any
		if (TG_BUFF_TOWN > 0 && TG_BUFF_TOWN != @_tg_town + 1) {
			callfunc "F_TG_RemoveBuff", TG_BUFF_TOWN - 1;
		}
		set $TG_TOP_ID[@_tg_town], getcharid(0);
		set $TG_TOP_NAME$[@_tg_town], strcharinfo(0);
		set $TG_TOP_ZEN[@_tg_town], @_tg_my_total;
		set TG_BUFF_TOWN, @_tg_town + 1;
		callfunc "F_TG_ApplyBuff", @_tg_town;
		mes "^663300[ Donate to " + @_tg_name$ + " ]^000000";
		mes "^00AA00You donated " + @_tg_amount + "z and became the Top Donor!^000000";
		mes "^00AA00Your buff has been applied!^000000";
		announce "[Town Governor] " + strcharinfo(0) + " is now the Top Donor of " + @_tg_name$ + "! (" + @_tg_my_total + "z)",0;
	} else {
		mes "^663300[ Donate to " + @_tg_name$ + " ]^000000";
		mes "^00AA00Thank you! You donated " + @_tg_amount + "z.^000000";
		mes "Your total: ^FF8800" + @_tg_my_total + "z^000000";
		mes "Top Donor needs: ^FF8800" + ($TG_TOP_ZEN[@_tg_town] - @_tg_my_total + 10000) + "z more^000000";
	}
	next;
	goto L_Main;

L_Board:
	mes "^663300[ " + @_tg_name$ + " Leaderboard ]^000000";
	if ($TG_TOP_ID[@_tg_town] > 0) {
		mes " ^FF8800#1 " + $TG_TOP_NAME$[@_tg_town] + "^000000 — " + $TG_TOP_ZEN[@_tg_town] + "z";
		mes " ^888888(Top Donor buff active)^000000";
	} else {
		mes "^888888No donations yet this week.^000000";
	}
	mes " ";
	mes "Weekly reset: ^888888Every Monday at 00:01^000000";
	next;
	goto L_Main;

// Weekly reset — every Monday 00:01
OnClock0001:
	if (gettime(4) != 1) end; // 1 = Monday
	set @_tg_i, 0;
	while (@_tg_i < 14) {
		set $TG_TOP_ID[@_tg_i], 0;
		set $TG_TOP_NAME$[@_tg_i], "";
		set $TG_TOP_ZEN[@_tg_i], 0;
		set @_tg_i, @_tg_i + 1;
	}
	announce "[Town Governor] Weekly donation reset! All town buffs and leaderboards have been cleared.",0;
	end;
}
