prontera,140,202,5	script	Resetadora	859,{

//===== Configurações: =======================================
set .@atrib, 25000;	// Reset de Atributos
set .@habil, 25000;	// Reset de Habilidades
set .@ambos, 50000;	// Reset de Ambos
//============================================================

	mes "[^010849===^000000 ^016B14"+StrNpcInfo(1)+"^000000 ^010849===^000000]";
	mes "Olá, posso resetar seus pontos.";
	mes "Qual você deseja resetar?";
	next;
	if (select ("Resetar Atributos ["+.@atrib+"z]:Resetar Habilidades ["+.@habil+"z]:Resetar Ambos ["+.@ambos+"z]:Cancelar") == 4) close;
	mes "[^010849===^000000 ^016B14"+StrNpcInfo(1)+"^000000 ^010849===^000000]";
	if (@menu == 1) callsub F_Reset, .@atrib, 1, 0;
	if (@menu == 2) callsub F_Reset, .@habil, 0, 1;
	if (@menu == 3) callsub F_Reset, .@ambos, 1, 1;
	mes "Pronto, resetado. Fale comigo mais vezes se precisar!";
	close;

	F_Reset:
		if (Zeny < getarg(0))
		{
			mes "Me desculpe, mas você não possui o Zeny necessário.";
			close;
		}
		set Zeny, Zeny - getarg(0);
		if (getarg(1)) resetstatus;
		if (getarg(2)) resetskill;
	return;
}
