	mes "Hello, adventurer! I can exchange your ^FF8800Poring Coins^000000 for useful items.";
	mes " ";
	mes "Your Poring Coins: ^FF8800" + countitem(7539) + "^000000";
	next;

	set @_pc_sel, select(
		"Consumables",
		"Wings & Teleport",
		"Battle Supplies",
		"Close"
	);

	if (@_pc_sel == 4) { close; }
	if (@_pc_sel == 1) goto L_Consumables;
	if (@_pc_sel == 2) goto L_Wings;

//-------------------------------

	mes "^663300[ Lydia — Consumables ]^000000";
	mes "Your Poring Coins: ^FF8800" + countitem(7539) + "^000000";
	next;
	set @_pc_sel, select(
		"Field Manual 100% — ^FF880040 PC^000000",
		"Bubble Gum — ^FF8800700 PC^000000",
		"LV10 Blessing Scroll — ^FF88002 PC^000000",
		"LV10 Agi Scroll — ^FF88002 PC^000000",

		"Regeneration Potion — ^FF880015 PC^000000",
		"Regen Potion x10 — ^FF8800150 PC^000000",

	if (@_pc_sel ==  5) { set @_pc_item, 12211; set @_pc_qty,  1; set @_pc_cost,  20; }
	if (@_pc_sel ==  6) { set @_pc_item, 12211; set @_pc_qty, 10; set @_pc_cost, 200; }
	if (@_pc_sel ==  7) { set @_pc_item, 12215; set @_pc_qty, 10; set @_pc_cost,  20; }

	if (@_pc_sel == 10) { set @_pc_item, 12209; set @_pc_qty,  1; set @_pc_cost,   3; }
	if (@_pc_sel == 11) { set @_pc_item, 14535; set @_pc_qty,  1; set @_pc_cost,  15; }
	if (@_pc_sel == 12) { set @_pc_item, 14535; set @_pc_qty, 10; set @_pc_cost, 150; }
	if (@_pc_sel == 13) { set @_pc_item, 14537; set @_pc_qty,  1; set @_pc_cost,  15; }
	if (@_pc_sel == 14) { set @_pc_item, 13720; set @_pc_qty,  1; set @_pc_cost, 150; }

		"Red Butterfly Wing x5 — ^FF880050 PC^000000",

		mes "^663300[ Lydia ]^000000";
		mes "^FF0000You don't have enough Poring Coins.^000000";
		mes "You need ^FF8800" + @_pc_cost + " PC^000000 but only have ^FF8800" + countitem(7539) + " PC^000000.";
		close;
	}
	delitem 7539, @_pc_cost;
	getitem @_pc_item, @_pc_qty;
	mes "^663300[ Lydia ]^000000";
	mes "^00AA00Exchange successful!^000000";
	mes "Remaining Poring Coins: ^FF8800" + countitem(7539) + "^000000";
