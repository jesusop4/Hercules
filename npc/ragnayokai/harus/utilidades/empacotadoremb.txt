function	script	Empacotador de Emblemas	{
	mes "[^010849===^000000 ^016B14"+StrNpcInfo(1)+"^000000 ^010849===^000000]";
	mes "Olá "+StrCharInfo(0)+"!";
	mes "Eu posso empacotar os seus emblemas em pacotes de ^016B141.000x^000000 unidades, deseja dar uma olhada?";
	next;
	switch(select("- Emblemas de ^016B14Valor^000000:- Emblemas de ^016B14Guerra^000000:- Emblemas de ^016B14Bravura^000000:Nenhum"))
	{
	case 1:
			close2;
			callshop "PacotaValor",0;
			end;
	case 2:
			close2;
			callshop "PacotaGuerra",0;
			end;
	case 3:
			close2;
			callshop "PacotaBravura",0;
			end;
	case 4: close;
	}
}
