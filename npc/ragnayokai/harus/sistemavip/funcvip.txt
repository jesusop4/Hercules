        mes "*^ED2020Não mude o seu equipamento.^000000";
        mes "*^ED2020Não mude seu stats de base.^000000";
        mes "^610B0BSe você não cumprir essas regras após o Emperium está quebrado vai ser expulso da Sala e você não vai ter o seu registro.^000000";
        if (@EmpMSP) { set .@a,select("Sair da Sala."); set .@a,.@a+1; }
        else { set .@a,select("Iniciar. [ 3 Seconds ]:Sair da Sala."); }
        if (.@a == 2) {
                if (readparam(bLuk) <= 1) { statusup2 bLuk,1; statusup2 bLuk,-1; } else { statusup2 bLuk,-1; statusup2 bLuk,1; }
                percentheal 0,100;
                warp "prontera",169,206;
                end;
        }
        close2;
        areaannounce "pvp_n_5-5",128,94,139,105,"Emperium irá aparecer ...",0;
        percentheal 0,-100;
        set MaxSp,1;
        set @EmpMSP,1;
        initnpctimer;
        end;

OnTimer1000:
        areaannounce "pvp_n_5-5",128,94,139,105,"3",0;
        end;

OnTimer2000:
        areaannounce "pvp_n_5-5",128,94,139,105,"2",0;
        end;

OnTimer3000:
        areaannounce "pvp_n_5-5",128,94,139,105,"1",0;
        end;

OnTimer4000:
        areaannounce "pvp_n_5-5",128,94,139,105,"GO!",0;
        monster "pvp_n_5-5",134,99,"Emperos Slave",$@EmpeMAV,1,strnpcinfo(0)+"::OnEDie";
        end;

OnTimer304000:
        areaannounce "pvp_n_5-5",128,94,139,105,"Limite de tempo excedido.",0;
        killmonster "pvp_n_5-5",strnpcinfo(0)+"::OnEDie";
        sleep2 2000;
        areawarp "pvp_n_5-5",128,94,139,105,"prontera",169,206;
        stopnpctimer;
        end;

OnEDie:
        stopnpctimer;
        if (getareausers ("pvp_n_5-5",128,94,139,105) == 0) end;
        if (MaxSp > 1) { warp "prontera",169,206; end; }
        if (readparam(bLuk) <= 1) { statusup2 bLuk,1; statusup2 bLuk,-1;} else { statusup2 bLuk,-1; statusup2 bLuk,1; }
        percentheal 0,100;

        callsub CS_Vars;

        set .@Min,.MyTime / 60000;
        set .MyTime,.MyTime - (.@Min * 60000);
        set .@Seg,.MyTime / 1000;
        set .MyTime,.MyTime - (.@Seg * 1000);

        areaannounce "pvp_n_5-5",128,94,139,105,"Seu Record é: "+.@Min+" Minuto"+(.@Min == 1 ? "":"s")+", "+.@Seg+" Segundo"+(.@Seg == 1 ? "":"s")+" e "+.MyTime+" Milissegundos.",0;
        sleep2 3000;
        areawarp "pvp_n_5-5",128,94,139,105,"prontera",169,206;
        end;
       
CS_Vars:
        set .MyTime,getnpctimer(0) - 4000;
        for (set .@a,1; .@a <= $@EmpeBMR; set .@a,.@a+1) {
                if (.MyTime < $EmpeBRec[.@a] || $EmpeBRec[.@a] == 0) {
                        for (set .@b,$@EmpeBMR; .@b >= .@a; set .@b,.@b-1) {
                                set $EmpeBRec[.@b],$EmpeBRec[(.@b - 1)];
                                set $EmpeBName$[.@b],$EmpeBName$[(.@b - 1)];
                        }
                        set $EmpeBRec[.@a],.MyTime;
                        set $EmpeBName$[.@a],strcharinfo(0);
                        return;
                }
        }
        return;
}
