	//Definir os mapas que será feita a contagem.
	setarray .mapas$[1],"prtg_cas01","arug_cas01","arug_cas03";
	set @qtd, 0;
	for( set .@i, 1; .@i <= getarraysize(.mapas$) - 1; set .@i, .@i + 1){
		set @qtd, @qtd + getMapGuildUsers (.mapas$[.@i],GetArg(0));
	}
	return @qtd;
}
